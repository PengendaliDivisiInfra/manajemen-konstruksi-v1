/**
 * Manajemen Konstruksi v1 — Backend Spreadsheet
 * Endpoint: Web App (Execute as: Me, Who has access: Anyone)
 */
const SHEET_NAMES = {
  master_resources      : 'master_resources',
  ahsp_headers          : 'ahsp_headers',
  ahsp_details_master   : 'ahsp_details_master',
  projects              : 'projects',
  project_ahsp_details  : 'project_ahsp_details',
  project_wbs           : 'project_wbs',
  progress              : 'progress',
  _settings             : '_settings'
};

function doPost(e) {
  try {
    const body = JSON.parse(e.postData.contents);
    const action = body.action;

    if (action === 'push') {
      const db = body.payload.db;
      const settings = body.payload.settings || {};
      writeAll_(db, settings);
      return json_({ ok: true, message: 'Semua tabel tersinkronisasi.' });
    }

    if (action === 'pull') {
      const db = readAll_();
      const settings = readSettings_();
      return json_({ ok: true, db, settings });
    }

    return json_({ ok: false, message: 'Action tidak dikenal: ' + action });
  } catch (err) {
    return json_({ ok: false, message: err.message });
  }
}

function doGet() {
  return json_({ ok: true, message: 'Manajemen Konstruksi v1 API aktif.' });
}

/* ---------- WRITE ---------- */
function writeAll_(db, settings) {
  Object.keys(SHEET_NAMES).forEach(key => {
    if (key === '_settings') return;
    const rows = db[key] || [];
    writeSheet_(SHEET_NAMES[key], rows);
  });
  writeSettings_(settings);
}

/* =====================================================================
   WRITE SHEET — dengan Auto-Styling Profesional
   ===================================================================== */
function writeSheet_(name, rows) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(name);
  if (!sh) sh = ss.insertSheet(name);

  // Bersihkan konten, format, dan banding lama
  sh.clear();
  sh.clearFormats();
  try { sh.getBandings().forEach(b => b.remove()); } catch(e){}

  if (!rows.length) {
    sh.getRange(1,1).setValue('(kosong)')
      .setFontStyle('italic').setFontColor('#888888');
    sh.setColumnWidth(1, 200);
    return;
  }

  // --- Kumpulkan seluruh header unik ---
  const headers = [];
  rows.forEach(r => Object.keys(r).forEach(k => {
    if (headers.indexOf(k) < 0) headers.push(k);
  }));
  headers.sort();

  // --- Bangun matriks nilai ---
  const values = [headers];
  rows.forEach(r => {
    values.push(headers.map(h => {
      const v = r[h];
      if (v === null || v === undefined) return '';
      if (typeof v === 'object') return JSON.stringify(v);
      return v;
    }));
  });

  // --- Tulis data mentah ---
  const range = sh.getRange(1, 1, values.length, headers.length);
  range.setValues(values);

  /* ============================================================
     STYLING PROFESIONAL
     ============================================================ */

  // 1. Header — biru tua korporat, teks putih, bold, tengah
  const headerRange = sh.getRange(1, 1, 1, headers.length);
  headerRange
    .setFontWeight('bold')
    .setFontColor('#ffffff')
    .setBackground('#1f4e79')
    .setHorizontalAlignment('center')
    .setVerticalAlignment('middle')
    .setFontFamily('Arial')
    .setFontSize(11);
  sh.setRowHeight(1, 34);
  sh.setFrozenRows(1);

  // 2. Border grid abu-abu tipis di seluruh area data
  const dataArea = sh.getRange(1, 1, values.length, headers.length);
  dataArea.setBorder(
    true, true, true, true, true, true,
    '#cccccc',
    SpreadsheetApp.BorderStyle.SOLID
  );
  dataArea.setFontFamily('Arial').setFontSize(10);
  dataArea.setVerticalAlignment('middle');

  // 3. Auto-resize kolom (dengan batas min/max)
  for (let c = 1; c <= headers.length; c++) {
    sh.autoResizeColumn(c);
    const w = sh.getColumnWidth(c);
    if (w > 320) sh.setColumnWidth(c, 320);
    if (w < 80)  sh.setColumnWidth(c, 80);
  }

  // 4. Number formatting berdasarkan nama kolom
  const hLower = headers.map(h => String(h).toLowerCase());
  const ccyPattern = '"Rp"#,##0.00';
  const decPattern = '0.000000';
  const numPattern = '#,##0.00';
  const intPattern = '#,##0';

  for (let c = 0; c < headers.length; c++) {
    const h = hLower[c];
    const colRange = sh.getRange(2, c + 1, values.length - 1, 1);
    const isNumeric = ['harga','total','nilai','biaya','varian','rp','ppn','margin','deviasi','subtotal','nilai_kontrak','harga_rab','harga_rap','harga_satuan','harga_satuan_rab','harga_satuan_rap','total_rab','total_rap']
      .some(kw => h.includes(kw));
    const isCoef = ['koefisien','faktor','kapasitas','persen','ratio','bucket','siklus']
      .some(kw => h.includes(kw));
    const isVolume = ['volume','qty','jumlah','vol_','koef']
      .some(kw => h.includes(kw));

    if (isCoef) {
      colRange.setNumberFormat(decPattern).setHorizontalAlignment('right');
    } else if (isNumeric) {
      colRange.setNumberFormat(ccyPattern).setHorizontalAlignment('right');
    } else if (isVolume && !isNaN(values[1] && values[1][c])) {
      colRange.setNumberFormat(numPattern).setHorizontalAlignment('right');
    } else if (!isNaN(values[1] && values[1][c])) {
      colRange.setNumberFormat(intPattern).setHorizontalAlignment('right');
    } else {
      colRange.setHorizontalAlignment('left');
    }
  }

  // 5. Baris selang-seling (banding) untuk readability
  try {
    const banding = dataArea.applyRowBanding(SpreadsheetApp.BandingTheme.LIGHT_GREY, false, false);
    if (banding) {
      banding.setHeaderRowColor('#1f4e79')
             .setFirstRowColor('#f3f7fb')
             .setSecondRowColor('#ffffff')
             .setFooterRowColor(null);
    }
  } catch(e) {
    // Fallback bila applyRowBanding gagal (sheet sudah punya banding)
    for (let i = 2; i <= values.length; i++) {
      if (i % 2 === 0) {
        sh.getRange(i, 1, 1, headers.length).setBackground('#f3f7fb');
      }
    }
  }

  // 6. Highlight kolom ID menjadi monospace kecil
  const idCols = headers.map((h, i) => /^(id|_id|project_id|ahsp_id|resource_id|wbs_id)$/i.test(h) ? i : -1)
    .filter(i => i >= 0);
  idCols.forEach(i => {
    sh.getRange(2, i + 1, values.length - 1, 1)
      .setFontFamily('Consolas')
      .setFontSize(9)
      .setFontColor('#666666');
  });
}

function writeSettings_(settings) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(SHEET_NAMES._settings);
  if (!sh) sh = ss.insertSheet(SHEET_NAMES._settings);
  sh.clear();

  // Header
  sh.getRange(1, 1, 1, 2).setValues([['key','value']])
    .setFontWeight('bold')
    .setBackground('#1f4e79')
    .setFontColor('#ffffff')
    .setHorizontalAlignment('center');
  sh.setFrozenRows(1);

  // Data
  const rows = Object.keys(settings).map(k => [k, settings[k]]);
  if (rows.length) {
    sh.getRange(2, 1, rows.length, 2).setValues(rows);
    sh.getRange(1, 1, rows.length + 1, 2).setBorder(true, true, true, true, true, true, '#cccccc', SpreadsheetApp.BorderStyle.SOLID);
  }
  sh.autoResizeColumn(1);
  sh.autoResizeColumn(2);
  sh.setColumnWidth(1, 180);
}

/* ---------- READ ---------- */
function readAll_() {
  const out = {};
  Object.keys(SHEET_NAMES).forEach(key => {
    if (key === '_settings') return;
    out[key] = readSheet_(SHEET_NAMES[key]);
  });
  return out;
}

function readSheet_(name) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sh = ss.getSheetByName(name);
  if (!sh) return [];
  const lastRow = sh.getLastRow();
  const lastCol = sh.getLastColumn();
  if (lastRow < 2 || lastCol < 1) return [];

  const values = sh.getRange(1, 1, lastRow, lastCol).getValues();
  const headers = values[0].map(h => String(h).trim()).filter(h => h !== '');
  const rows = [];

  for (let i = 1; i < values.length; i++) {
    const obj = {};
    let hasVal = false;
    headers.forEach((h, j) => {
      let v = values[i][j];
      if (v instanceof Date) v = v.toISOString().slice(0,10);
      // konversi angka otomatis
      if (typeof v === 'string' && v !== '' && !isNaN(v) && v.trim() !== '' && /^-?\d+(\.\d+)?$/.test(v.trim())) {
        v = parseFloat(v);
      }
      // parse JSON bila perlu
      if (typeof v === 'string' && (v.startsWith('{') || v.startsWith('['))) {
        try { v = JSON.parse(v); } catch(e){}
      }
      obj[h] = v;
      if (v !== '' && v !== null) hasVal = true;
    });
    if (hasVal) rows.push(obj);
  }
  return rows;
}

function readSettings_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sh = ss.getSheetByName(SHEET_NAMES._settings);
  if (!sh) return {};
  const lastRow = sh.getLastRow();
  if (lastRow < 2) return {};
  const values = sh.getRange(2, 1, lastRow-1, 2).getValues();
  const obj = {};
  values.forEach(([k,v]) => { if (k) obj[k] = v; });
  return obj;
}

/* ---------- HELPER ---------- */
function json_(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
